<objective>
Wave 2 GREEN: implement the full execution engine to make all test_runs.py tests pass, completing Phase 2.

Two critical architectural invariants:

1. Persist-before-dispatch: every state transition is committed to SQLite BEFORE the outbound HTTP
   call. Sequence: (a) set step status=RUNNING, commit; (b) POST to execute_url; (c) set result,
   commit. If the process crashes between (a) and (c), the DB records RUNNING — a zombie detection
   sweep (v2) can recover it.

2. Background task session ownership: execute_run creates its own AsyncSession via AsyncSessionLocal()
   — it cannot share the request session, which is closed when the 202 response is sent. For test
   isolation, executor.py exposes AsyncSessionLocal as a module-level name so tests can replace it
   with a session factory pointing to the test database.
</objective>

    from uuid import uuid4
    from sqlalchemy import select
    from sqlalchemy.ext.asyncio import AsyncSession
    from app.core.database import _write_lock
    from app.runs.models import WorkflowRun, WorkflowStepExecution


    class RunRepo:
        def __init__(self, session: AsyncSession) -> None:
            self._session = session

        async def create(self, workflow_def_id: str, input_payload: dict | None) -> WorkflowRun:
            async with _write_lock:
                run = WorkflowRun(
                    id=str(uuid4()),
                    workflow_def_id=workflow_def_id,
                    status="PENDING",
                    input_payload=json.dumps(input_payload) if input_payload else None,
                )
                self._session.add(run)
                await self._session.commit()
                return run

        async def get_by_id(self, run_id: str) -> WorkflowRun | None:
            result = await self._session.execute(
                select(WorkflowRun).where(WorkflowRun.id == run_id)
            )
            return result.scalar_one_or_none()

        async def list_steps_by_run_id(self, run_id: str) -> list[WorkflowStepExecution]:
            result = await self._session.execute(
                select(WorkflowStepExecution)
                .where(WorkflowStepExecution.run_id == run_id)
                .order_by(WorkflowStepExecution.step_index)
            )
            return list(result.scalars().all())

    import json
    from datetime import datetime
    from uuid import uuid4

    import httpx
    from sqlalchemy import select

    # Module-level import — patchable by tests via:
    #   import app.runs.executor as m; m.AsyncSessionLocal = test_factory
    from app.core.database import AsyncSessionLocal
    from app.core.database import _write_lock
    from app.runs.models import WorkflowRun, WorkflowStepExecution
    from app.workflows.models import WorkflowDefinition


    async def execute_run(run_id: str, workflow_def_id: str) -> None:
        """
        Background task: drive a workflow run from PENDING to terminal state.

        Creates its own AsyncSession — cannot reuse the request session (closed after 202 response).
        Uses the module-level AsyncSessionLocal so tests can patch it to the test engine.

        Persist-before-dispatch: every status transition committed to DB BEFORE HTTP call.
        """
        # Use module-level name so tests can patch app.runs.executor.AsyncSessionLocal
        import app.runs.executor as _self
        session_factory = _self.AsyncSessionLocal

        async with session_factory() as session:
            run_result = await session.execute(
                select(WorkflowRun).where(WorkflowRun.id == run_id)
            )
            run = run_result.scalar_one()

            wf_result = await session.execute(
                select(WorkflowDefinition).where(WorkflowDefinition.id == workflow_def_id)
            )
            wf = wf_result.scalar_one()

            # Transition run PENDING → RUNNING — persist before dispatch
            async with _write_lock:
                run.status = "RUNNING"
                run.started_at = datetime.utcnow()
                await session.commit()

            async with httpx.AsyncClient() as http_client:
                for step in wf.steps:
                    # Persist step as RUNNING before HTTP call
                    step_exec = WorkflowStepExecution(
                        id=str(uuid4()),
                        run_id=run_id,
                        step_index=step.step_order,
                        step_name=step.name,
                        status="RUNNING",
                        started_at=datetime.utcnow(),
                    )
                    async with _write_lock:
                        session.add(step_exec)
                        run.current_step_index = step.step_order
                        await session.commit()

                    timeout = step.timeout_seconds or 30
                    try:
                        response = await http_client.post(
                            step.execute_url,
                            json=json.loads(run.input_payload) if run.input_payload else {},
                            timeout=httpx.Timeout(timeout),
                        )
                        if 200 <= response.status_code < 300:
                            async with _write_lock:
                                step_exec.status = "COMPLETED"
                                step_exec.http_status_code = response.status_code
                                step_exec.completed_at = datetime.utcnow()
                                await session.commit()
                        else:
                            async with _write_lock:
                                step_exec.status = "FAILED"
                                step_exec.http_status_code = response.status_code
                                step_exec.error_detail = f"HTTP {response.status_code}"
                                step_exec.completed_at = datetime.utcnow()
                                run.status = "FAILED"
                                run.error_detail = (
                                    f"Step '{step.name}' failed: HTTP {response.status_code}"
                                )
                                run.completed_at = datetime.utcnow()
                                await session.commit()
                            return
                    except httpx.TimeoutException:
                        async with _write_lock:
                            step_exec.status = "FAILED"
                            step_exec.error_detail = f"Timeout after {timeout}s"
                            step_exec.completed_at = datetime.utcnow()
                            run.status = "FAILED"
                            run.error_detail = f"Step '{step.name}' timed out after {timeout}s"
                            run.completed_at = datetime.utcnow()
                            await session.commit()
                        return

            # All steps completed successfully
            async with _write_lock:
                run.status = "COMPLETED"
                run.completed_at = datetime.utcnow()
                await session.commit()
    ```

    from typing import Annotated
    from fastapi import APIRouter, BackgroundTasks, Depends
    from fastapi.exceptions import HTTPException
    from sqlalchemy import select
    from sqlalchemy.ext.asyncio import AsyncSession

    from app.core.auth import APIKey
    from app.core.database import get_db
    from app.runs.executor import execute_run
    from app.runs.repository import RunRepo
    from app.runs.schemas import RunResponse, StartRunRequest, StepExecutionResponse
    from app.workflows.models import WorkflowDefinition

    router = APIRouter(tags=["runs"])
    DBSession = Annotated[AsyncSession, Depends(get_db)]


    @router.post("/workflows/{workflow_name}/runs", status_code=202)
    async def start_run(
        workflow_name: str,
        body: StartRunRequest,
        background_tasks: BackgroundTasks,
        db: DBSession,
        _: APIKey,
    ) -> dict:
        wf_result = await db.execute(
            select(WorkflowDefinition).where(WorkflowDefinition.id == workflow_name)
        )
        if wf_result.scalar_one_or_none() is None:
            raise HTTPException(status_code=404, detail=f"Workflow '{workflow_name}' not found")
        repo = RunRepo(db)
        run = await repo.create(workflow_name, body.input)
        background_tasks.add_task(execute_run, run.id, workflow_name)
        return {"run_id": run.id}


    @router.get("/runs/{run_id}", response_model=RunResponse)
    async def get_run(run_id: str, db: DBSession, _: APIKey) -> RunResponse:
        run = await RunRepo(db).get_by_id(run_id)
        if run is None:
            raise HTTPException(status_code=404, detail=f"Run '{run_id}' not found")
        return RunResponse.model_validate(run)


    @router.get("/runs/{run_id}/steps", response_model=list[StepExecutionResponse])
    async def get_run_steps(run_id: str, db: DBSession, _: APIKey) -> list[StepExecutionResponse]:
        repo = RunRepo(db)
        if await repo.get_by_id(run_id) is None:
            raise HTTPException(status_code=404, detail=f"Run '{run_id}' not found")
        steps = await repo.list_steps_by_run_id(run_id)
        return [StepExecutionResponse.model_validate(s) for s in steps]
    ```
