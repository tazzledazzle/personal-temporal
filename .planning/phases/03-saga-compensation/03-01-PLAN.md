    """
    COMP-01, COMP-02: Tests for Saga compensation (reverse-order HTTP rollback).

    When any step fails, the executor calls each previously-completed step's compensate_url
    in reverse-completion order (LIFO). Compensation continues even if individual calls fail.
    Per-step outcomes are recorded; final run status reflects overall compensation result.

    Synchronization: BackgroundTasks (including compensation) complete before
    await client.post() returns — ASGITransport closes the ASGI scope only after all
    background tasks finish.
    """

    import respx
    import httpx
    from app.runs.compensator import CompensationCoordinator  # RED — does not exist yet

    # 3-step workflow: step-a and step-b have compensate_urls; step-c does not
    COMP_WORKFLOW = {
        "name": "comp-wf",
        "steps": [
            {
                "name": "step-a",
                "execute_url": "http://mock.test/exec-a",
                "compensate_url": "http://mock.test/comp-a",
            },
            {
                "name": "step-b",
                "execute_url": "http://mock.test/exec-b",
                "compensate_url": "http://mock.test/comp-b",
            },
            {
                "name": "step-c",
                "execute_url": "http://mock.test/exec-c",
                # no compensate_url — silently skipped
            },
        ],
    }


    async def test_compensation_called_on_step_failure(client):
        """COMP-01: When a step fails, compensate all previously-completed steps."""
        await client.post("/workflows", json=COMP_WORKFLOW)
        async with respx.mock:
            respx.post("http://mock.test/exec-a").mock(return_value=httpx.Response(200))
            respx.post("http://mock.test/exec-b").mock(return_value=httpx.Response(500))
            respx.post("http://mock.test/comp-a").mock(return_value=httpx.Response(200))
            response = await client.post("/workflows/comp-wf/runs", json={"input": {}})
            run_id = response.json()["run_id"]
            steps_response = await client.get(f"/runs/{run_id}/steps")
        body = steps_response.json()
        step_a = next(s for s in body if s["step_name"] == "step-a")
        assert step_a["compensation_status"] == "COMPLETED", (
            f"Expected step-a compensation_status=COMPLETED, got {step_a['compensation_status']}"
        )
        assert step_a["compensated_at"] is not None, "Expected step-a compensated_at to be set"


    async def test_compensation_lifo_order(client):
        """COMP-01: Compensation calls happen in reverse-completion order (LIFO)."""
        await client.post("/workflows", json=COMP_WORKFLOW)
        call_order = []

        async def record_comp_b(request):
            call_order.append("comp-b")
            return httpx.Response(200)

        async def record_comp_a(request):
            call_order.append("comp-a")
            return httpx.Response(200)

        async with respx.mock:
            respx.post("http://mock.test/exec-a").mock(return_value=httpx.Response(200))
            respx.post("http://mock.test/exec-b").mock(return_value=httpx.Response(200))
            respx.post("http://mock.test/exec-c").mock(return_value=httpx.Response(500))
            respx.post("http://mock.test/comp-b").mock(side_effect=record_comp_b)
            respx.post("http://mock.test/comp-a").mock(side_effect=record_comp_a)
            await client.post("/workflows/comp-wf/runs", json={"input": {}})
            # Compensation completes before post() returns

        assert call_order == ["comp-b", "comp-a"], (
            f"Expected LIFO order [comp-b, comp-a], got {call_order}"
        )


    async def test_compensation_continues_on_failure(client):
        """COMP-01: Compensation chain continues even when an individual compensate call fails."""
        await client.post("/workflows", json=COMP_WORKFLOW)
        async with respx.mock:
            respx.post("http://mock.test/exec-a").mock(return_value=httpx.Response(200))
            respx.post("http://mock.test/exec-b").mock(return_value=httpx.Response(200))
            respx.post("http://mock.test/exec-c").mock(return_value=httpx.Response(500))
            respx.post("http://mock.test/comp-b").mock(return_value=httpx.Response(500))  # fails
            respx.post("http://mock.test/comp-a").mock(return_value=httpx.Response(200))  # succeeds
            response = await client.post("/workflows/comp-wf/runs", json={"input": {}})
            run_id = response.json()["run_id"]
            steps_response = await client.get(f"/runs/{run_id}/steps")
        body = steps_response.json()
        step_a = next(s for s in body if s["step_name"] == "step-a")
        step_b = next(s for s in body if s["step_name"] == "step-b")
        assert step_b["compensation_status"] == "FAILED", "step-b compensation should be FAILED"
        assert step_b["compensation_error"] is not None, "step-b should have compensation_error"
        assert step_a["compensation_status"] == "COMPLETED", "step-a should still be compensated"
        assert step_a["compensated_at"] is not None


    async def test_compensation_skips_steps_without_compensate_url(client):
        """COMP-01: Steps without compensate_url are silently skipped."""
        await client.post("/workflows", json=COMP_WORKFLOW)
        async with respx.mock:
            respx.post("http://mock.test/exec-a").mock(return_value=httpx.Response(200))
            respx.post("http://mock.test/exec-b").mock(return_value=httpx.Response(200))
            respx.post("http://mock.test/exec-c").mock(return_value=httpx.Response(500))
            respx.post("http://mock.test/comp-b").mock(return_value=httpx.Response(200))
            respx.post("http://mock.test/comp-a").mock(return_value=httpx.Response(200))
            response = await client.post("/workflows/comp-wf/runs", json={"input": {}})
            run_id = response.json()["run_id"]
            steps_response = await client.get(f"/runs/{run_id}/steps")
        body = steps_response.json()
        # step-c completed before failing — it has no compensate_url so its compensation_status is None
        step_c = next((s for s in body if s["step_name"] == "step-c"), None)
        # step-c FAILED (the step that triggered compensation), not COMPLETED, so it won't be compensated
        # step-b and step-a are COMPLETED — both get compensated
        step_b = next(s for s in body if s["step_name"] == "step-b")
        step_a = next(s for s in body if s["step_name"] == "step-a")
        assert step_b["compensation_status"] == "COMPLETED"
        assert step_a["compensation_status"] == "COMPLETED"


    async def test_run_status_compensated_when_all_succeed(client):
        """COMP-02: run.status == COMPENSATED when all compensation calls succeed."""
        await client.post("/workflows", json=COMP_WORKFLOW)
        async with respx.mock:

    async def test_run_status_compensation_failed_on_partial(client):

        )
